GSAP Parallax Script
gsap.registerPlugin();

const hero = document.getElementById('hero');
const bg = document.getElementById('bg');

// Set your image
bg.style.backgroundImage = "url('https://digitalassets.tesla.com/tesla-contents/image/upload/f_auto,q_auto/Model-Y-Hero.jpg')";

// Parallax on scroll
window.addEventListener('scroll', () => {
  const scrollY = window.scrollY;
  gsap.to(bg, {
    y: scrollY * 0.5,           // Adjust speed (0.3 - 0.6 feels natural)
    duration: 0.5,
    ease: "power1.out"
  });
});

// Optional: Fade in content
gsap.from(".hero-content", {
  y: 100,
  opacity: 0,
  duration: 1.2,
  delay: 0.5,
  ease: "power2.out"
});
